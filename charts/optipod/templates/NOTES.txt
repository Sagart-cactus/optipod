📋 Quick Start:

1. Create an OptimizationPolicy that applies to all workloads:

cat <<EOF | kubectl apply -f -
apiVersion: optipod.optipod.io/v1alpha1
kind: OptimizationPolicy
metadata:
  name: all-workloads
  namespace: {{ include "optipod.namespace" . }}
spec:
  mode: Auto
  selector:
    namespaceSelector:
      matchExpressions:
        - key: kubernetes.io/metadata.name
          operator: NotIn
          values:
            - kube-system
            - kube-public
            - kube-node-lease
  metricsConfig:
    provider: prometheus
    rollingWindow: 24h
    percentile: P90
    safetyFactor: 1.2
  resourceBounds:
    cpu:
      min: "100m"
      max: "4000m"
    memory:
      min: "128Mi"
      max: "8Gi"
  updateStrategy:
    allowInPlaceResize: true
    allowRecreate: false
    updateRequestsOnly: true
    useServerSideApply: true
  reconciliationInterval: 5m
EOF

2. Generate HTML recommendation report:

   curl -sL https://raw.githubusercontent.com/optipod/optipod/main/scripts/optipod-recommendation-report.sh | bash -s -- --output html --file report.html
   open report.html
